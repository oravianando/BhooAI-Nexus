Boot a backend with the security stack

The minimum runnable server: health, CSRF, and an echo route.

import { loadConfig, Router, NexusServer, bodyParser } from '@bhooai/nexus-core';
import { cors, csrf, securityHeaders, rateLimit, issueCsrfToken } from '@bhooai/nexus-auth';
import { Logger } from '@bhooai/nexus-telemetry';

const config = await loadConfig({ root: process.cwd() });
const log = new Logger({ level: config.logging.level, console: true });
const trustedOrigins = [
  `http://localhost:${config.server.port}`,
  `http://127.0.0.1:${config.server.port}`,
   'http://localhost:3000',
];

const router = new Router();
router.get('/health', (ctx) => ctx.json({ status: 'ok' }));
router.get('/csrf-token', (ctx) => ctx.json({ token: issueCsrfToken(ctx, { trustedOrigins }) }));
router.post('/echo', (ctx) => ctx.json({ received: ctx.body }));

const server = new NexusServer({
  router,
  middleware: [
    securityHeaders(),
    cors({ origin: true, credentials: true }),
    bodyParser(config.server.bodyLimit),
    csrf({ trustedOrigins }),
    rateLimit({ windowMs: 60_000, max: 300 }),
  ],
});
await server.listen(config.server.port, config.server.host);
CSRF POST /echo requires an Origin header matching a trusted origin and an x-csrf-token matching the nexus_csrf cookie from GET /csrf-token.

Upload a file to /uploads

The core upload helper stores multipart files under the project-local uploads/ directory and returns generated public filenames. Keep the static middleware before the body parser so GETs do not consume a request body.

import { join } from 'node:path';
import { registerUploadRoutes, serveStatic } from '@bhooai/nexus-core';

const uploadsDir = join(process.cwd(), config.uploads.dir);
registerUploadRoutes(router, {
  directory: uploadsDir,
  path: '/uploads',
  maxFileSize: config.uploads.maxFileSize,
  maxFiles: config.uploads.maxFiles,
  allowedTypes: ['image/png', 'image/jpeg'],
});

const server = new NexusServer({
  router,
  middleware: [
    serveStatic(uploadsDir, { prefix: '/uploads' }),
    bodyParser(config.server.bodyLimit),
    csrf({ trustedOrigins }),
  ],
});
// Browser: do not set Content-Type manually; FormData supplies the boundary.
const csrf = await fetch('/csrf-token', { credentials: 'include' }).then((r) => r.json());
const form = new FormData();
form.append('file', input.files[0]);
const result = await fetch('/uploads', {
  method: 'POST', body: form, credentials: 'include',
  headers: { 'x-csrf-token': csrf.token },
}).then((r) => r.json());
Limits The default request envelope is 12 MiB and the default individual file limit is 10 MiB. Configure server.bodyLimit, uploads.maxFileSize, uploads.maxFiles, and uploads.allowedTypes together for larger or restricted uploads.

Define an ODM model and query

import { connect, model, Schema, transaction } from '@bhooai/nexus-data';

connect('mongodb://localhost:27017/app');

const Post = model('Post', new Schema({
  title:   { type: String, required: true, index: true },
  author:  { type: 'ObjectId', ref: 'User', required: true },
  tags:    { type: [String], default: [] },
  views:   { type: Number, default: 0, immutable: true // can't be set after create },
  live:    { type: Boolean, default: true },
}, { timestamps: true }));

// chainable + thenable query
const hot = await Post.find({ live: true })
  .where('views').gt(100)
  .sort('-createdAt')
  .limit(10)
  .populate('author', 'name email')
  .lean();

// a transaction
await transaction(async (session) => {
  const post = await Post.create([{ title: 'Hello', author: userId }], { session });
  await Post.updateOne({ _id: post[0]._id }, { $push: { tags: 'intro' } }, { session });
});

Define a GraphQL subgraph

import { defineSubgraph, createGateway, graphqlHttpHandler } from '@bhooai/nexus-graphql';
import { model } from '@bhooai/nexus-data';

const Todo = model('Todo', /* schema */);

const subgraph = defineSubgraph({
  typeDefs: `
    type Todo @key(fields: "id") { id: ID! title: String! done: Boolean! }
    type Query { todos: [Todo!]! todo(id: ID!): Todo }
  `,
  resolvers: {
    Query: {
      todos: () => Todo.find().lean(),
      todo: (_p, { id }) => Todo.findById(id).lean(),
    },
    Todo: {
      __resolveReference: (rep) => Todo.findById(rep.id).lean(),
    },
  },
});

const gateway = createGateway({ subgraph });
router.post('/graphql', graphqlHttpHandler({ gateway, context: async (ctx) => ({ request: ctx }) }));

Auth: register, login, protect a route

import { AuthService, MemorySessionStore, authToken, requireRole, setAuthCookies } from '@bhooai/nexus-auth';

const auth = new AuthService({ secret: config.auth.jwt.secret, accessTtl: '15m', refreshTtl: '30d' }, new MemorySessionStore());

router.post('/login', async (ctx) => {
  // verify password, load user…
  const pair = await auth.login({ userId: String(user._id), roles: user.roles });
  setAuthCookies(ctx, pair, { accessTokenName: config.auth.cookieName, refreshTokenName: config.auth.refreshCookieName });
  ctx.json({ accessToken: pair.accessToken });
});

// protected route — middleware is the 3rd arg as an array, runs before the handler
router.get('/me', (ctx) => ctx.json({ id: ctx.state.user.id, roles: ctx.state.user.roles }),
  [authToken(auth, { cookieName: config.auth.cookieName, allowCookie: true }), requireRole('user')]);

Create a payment order

import { createPayments } from '@bhooai/nexus-payments';

 const payments = createPayments(config.payments); // keys in .env activate providers

router.post('/payments/order', async (ctx) => {
  const { provider, amount, currency } = ctx.body;
  const gateway = payments.get(provider); // throws when disabled
  const order = await gateway.createOrder({ amount, currency });
  ctx.json(order); // redirect or use the returned provider order id
}, [authToken(auth, { cookieName: config.auth.cookieName, allowCookie: true })]);

Send a templated email

Generate a self-signed cert + CSR

import { generateKeyPair, createSelfSignedCertificate, createCsr } from '@bhooai/nexus-crypto';

const kp = generateKeyPair('rsa', { modulusLength: 2048 });
const cert = createSelfSignedCertificate({ keyPair: kp, commonName: 'localhost' });
const csr = createCsr({ keyPair: kp, commonName: 'localhost', organization: 'BhooAI' });
console.log(cert.pem, kp.pem.private, csr.pem);

Stream an AI chat (Node proxy re-emits SSE)

import { AiClient } from '@bhooai/nexus-ai-client';
const ai = new AiClient({ serverUrl: config.ai.serverUrl, timeoutMs: config.ai.timeoutMs });

router.post('/ai/chat', async (ctx) => {
  ctx.res.writeHead(200, { 'content-type': 'text/event-stream' });
  for await (const chunk of ai.chatStream(ctx.body)) {
    ctx.res.write(`data: ${JSON.stringify(chunk)}\n\n`);
  }
  ctx.res.write('data: [DONE]\n\n');
  ctx.res.end();
});

A plugin manifest

// plugins/analytics/plugin.json
{
  "name": "analytics",
  "version": "1.0.0",
  "entry": "index.js",
  "mode": "trusted",
  "capabilities": ["http", "events", "admin"],
  "hooks": ["init", "start", "stop"],
  "dependencies": []
}
// plugins/analytics/index.js
export default {
  async start(ctx) {
    ctx.http.addRoute('GET', '/analytics/hit', (r) => r.json({ ok: true }));
    ctx.admin.registerPage({ id: 'analytics', label: 'Analytics', group: 'Insights' });
    ctx.events.on('user:created', () => ctx.log.info('new user'));
  },
};
